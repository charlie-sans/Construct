📝 EXAMPLE CODE WITH SEMANTIC HIGHLIGHTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// This is a comment                          ← GRAY
                                              
fn main =>                                    ← fn=PURPLE, main=WHITE
  let x = 42                                  ← let=PURPLE, x=WHITE, 42=CYAN
  dump(x)                                     ← dump=ORANGE, x=WHITE
  print("Hello, World!")                      ← print=ORANGE, string=GREEN
  nl()                                        ← nl=ORANGE
  
  let Result = "success"                      ← let=PURPLE, Result=BLUE, str=GREEN
  let len_val = len(Result)                   ← len=ORANGE
  
  if x > 0 =>                                 ← if=PURPLE, >=PURPLE
    dump(concat("Value: ", toStr(x)))         ← dump/concat/toStr=ORANGE
  else
    dump("Zero or negative")                  ← dump=ORANGE
  
  match x =>                                  ← match=PURPLE
    | 42 => print("Answer!")                  ← |=PURPLE, 42=CYAN, print=ORANGE
    | _ => print("Other")                     ← |=PURPLE, print=ORANGE

🎯 HOW IT WORKS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. You open a .cst file
   ↓
2. Extension activates semantic token provider
   ↓
3. Document content processed line by line
   ↓
4. Each character sequence classified:
   - Comments (// and /* */)
   - Strings (quoted text)
   - Numbers (integers and floats)
   - Keywords (from keyword list)
   - Built-ins (from stdlib list)
   - Operators (+, -, *, ==, etc.)
   - Delimiters ((, ), [, ], {, }, etc.)
   - Everything else → variable
   ↓
5. Tokens assigned types and modifiers
   ↓
6. VS Code theme applies colors
   ↓
7. Beautiful syntax highlighting in editor!

🎨 CUSTOMIZATION EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Add to VS Code settings (settings.json):

{
  "editor.semanticTokenColorCustomizations": {
    "rules": {
      "builtin:*": "#FF8C00",              // Orange for built-ins
      "keyword": "#6A5ACD",                // Purple for keywords
      "type": "#00CED1",                   // Cyan for types
      "variable.readonly": "#008000"       // Green for readonly
    }
  }
}
