OUTPUT FUNCTIONS (3)
  dump(value)       → Output any value to stdout
  print(fmt, ...)   → Formatted string output (printf-style)
  nl()              → Print newline

STRING FUNCTIONS (5)
  len(s)                    → Get string length
  concat(a, b)              → Concatenate two strings
  eq(a, b)                  → Check string equality
  substr(s, start, length)  → Extract substring
  indexOf(s, char)          → Find character position

TYPE CONVERSION (4)
  toStr(value)   → Convert to string
  toInt(s)       → Convert to integer
  toFloat(s)     → Convert to float
  toBool(s)      → Convert to boolean

MATH FUNCTIONS (5)
  abs(x)       → Absolute value
  sqrt(x)      → Square root
  pow(b, e)    → Base to power
  max(a, b)    → Maximum
  min(a, b)    → Minimum

KEYWORDS (9)

✨ USAGE EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Basic Output:
  dump(42)
  dump(3.14)
  dump("Hello")
  nl()

String Operations:
  len("hello")                  → 5
  concat("hello", " world")     → "hello world"
  eq("a", "b")                  → false
  substr("hello", 1, 3)         → "ell"
  indexOf("hello", 'l')         → 2

Type Conversion:
  toStr(42)        → "42"
  toInt("123")     → 123
  toFloat("3.14")  → 3.14
  toBool("true")   → true

Math Operations:
  abs(-5)          → 5
  sqrt(16)         → 4.0
  pow(2, 8)        → 256
  max(10, 20)      → 20
  min(10, 20)      → 10

Complete Program:
  fn main =>
    let x = 42
    dump(x)
    nl()
    dump(concat("Value: ", toStr(x)))
    nl()

1. Build the compiler:
   $ meson setup build && cd build && ninja

2. Test the feature:
   $ ./construct --list-stdlib

3. In VS Code:
   - Open a .cst file
   - Start typing: dump, len, print, etc.
   - See auto-completions with parameters
   - Press Tab/Enter to insert

4. Try these keystrokes:
   d + u + m  → dump(${1:value})
   l + e + n  → len(${1:s})
   t + o + S  → toStr(${1:value})
   p + o + w  → pow(${1:base}, ${2:exp})
